cmake_minimum_required(VERSION 3.22)
project(frotz-zero VERSION 0.8.1 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_SKIP_RPATH TRUE)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/verify_vendored.py COMMAND_ERROR_IS_FATAL ANY)
if(CMAKE_CROSSCOMPILING)
  # The official SDK contains runtime libraries; its pkg-config files refer to
  # private static-link dependencies whose development packages are not shipped.
  # We dynamically link the verified target libraries directly instead.
  add_library(PkgConfig::DEPS INTERFACE IMPORTED)
  set_property(TARGET PkgConfig::DEPS PROPERTY INTERFACE_INCLUDE_DIRECTORIES
    "${CMAKE_SYSROOT}/usr/include/SDL2;${CMAKE_SYSROOT}/usr/include/freetype2;${CMAKE_SYSROOT}/usr/include/aarch64-linux-gnu")
  foreach(name SDL2 SDL2_ttf SDL2_mixer freetype png z xkbcommon crypto curl)
    find_library(FZ_${name} NAMES ${name} PATHS "${CMAKE_SYSROOT}/usr/lib/aarch64-linux-gnu" NO_DEFAULT_PATH REQUIRED)
    set_property(TARGET PkgConfig::DEPS APPEND PROPERTY INTERFACE_LINK_LIBRARIES "${FZ_${name}}")
  endforeach()
  find_library(JPEG_LIBRARY NAMES jpeg PATHS "${CMAKE_SYSROOT}/usr/lib/aarch64-linux-gnu" NO_DEFAULT_PATH REQUIRED)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(DEPS REQUIRED IMPORTED_TARGET sdl2 SDL2_ttf SDL2_mixer freetype2 libpng zlib xkbcommon openssl libcurl)
endif()
find_path(JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
find_package(Threads REQUIRED)
find_package(JPEG REQUIRED)
find_path(ARCHIVE_INCLUDE_DIR archive.h REQUIRED)
find_path(XML_INCLUDE_DIR libxml/xmlreader.h PATH_SUFFIXES libxml2 REQUIRED)
if(CMAKE_CROSSCOMPILING)
  find_library(ARCHIVE_LIBRARY NAMES archive PATHS "${CMAKE_SYSROOT}/usr/lib/aarch64-linux-gnu" NO_DEFAULT_PATH REQUIRED)
  find_library(XML_LIBRARY NAMES xml2 PATHS "${CMAKE_SYSROOT}/usr/lib/aarch64-linux-gnu" NO_DEFAULT_PATH REQUIRED)
else()
  find_library(ARCHIVE_LIBRARY NAMES archive REQUIRED)
  find_library(XML_LIBRARY NAMES xml2 REQUIRED)
endif()
set(ENGINE_DIR ${CMAKE_CURRENT_BINARY_DIR}/engine)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS scripts/prepare_engine.py)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/prepare_engine.py ${ENGINE_DIR} COMMAND_ERROR_IS_FATAL ANY)
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS ${ENGINE_DIR}/src/common/*.c)
file(GLOB SDL_SOURCES CONFIGURE_DEPENDS ${ENGINE_DIR}/src/sdl/*.c)
file(GLOB BLORB_SOURCES CONFIGURE_DEPENDS ${ENGINE_DIR}/src/blorb/*.c)
add_library(frotz-engine STATIC ${CORE_SOURCES} ${SDL_SOURCES} ${BLORB_SOURCES})
target_include_directories(frotz-engine PRIVATE src ${ENGINE_DIR}/src/common ${ENGINE_DIR}/src/sdl)
target_compile_definitions(frotz-engine PRIVATE _DEFAULT_SOURCE)
target_compile_options(frotz-engine PRIVATE -Wno-unused-function -Wno-unused-variable)
target_link_libraries(frotz-engine PUBLIC PkgConfig::DEPS JPEG::JPEG m)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS assets/story-catalog.json scripts/prepare_catalog.py)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/prepare_catalog.py ${CMAKE_CURRENT_BINARY_DIR}/generated/story_catalog.inc COMMAND_ERROR_IS_FATAL ANY)
add_library(frotz-model STATIC src/model.cpp src/catalog.cpp)
target_include_directories(frotz-model PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_include_directories(frotz-model PUBLIC src)
target_link_libraries(frotz-model PUBLIC PkgConfig::DEPS)
add_library(frotz-online STATIC src/online.cpp src/archive.cpp)
target_compile_definitions(frotz-online PRIVATE FROTZ_ZERO_VERSION="${PROJECT_VERSION}")
target_include_directories(frotz-online PRIVATE ${JSON_INCLUDE_DIR} ${ARCHIVE_INCLUDE_DIR} ${XML_INCLUDE_DIR})
target_link_libraries(frotz-online PUBLIC frotz-model ${ARCHIVE_LIBRARY} ${XML_LIBRARY})
add_library(qrcodegen STATIC third_party/qrcodegen/qrcodegen.c)
target_include_directories(qrcodegen PUBLIC third_party/qrcodegen)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS scripts/prepare_ui.py
  assets/brand-24.png assets/brand-40.png assets/brand-96.png app-builder.json
  third_party/frotz/AUTHORS assets/code-authors.txt THIRD_PARTY_NOTICES.md LICENSE)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/prepare_ui.py ${CMAKE_CURRENT_BINARY_DIR}/generated COMMAND_ERROR_IS_FATAL ANY)
add_executable(frotz-zero src/main.cpp src/platform.cpp src/library_ui.cpp src/online_ui.cpp src/engine.cpp src/save_ui.cpp src/artwork.cpp src/credits.cpp src/help.cpp)
target_include_directories(frotz-zero PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_compile_definitions(frotz-zero PRIVATE FROTZ_ZERO_VERSION="${PROJECT_VERSION}")
target_link_libraries(frotz-zero PRIVATE frotz-engine frotz-online qrcodegen Threads::Threads)
target_compile_options(frotz-zero PRIVATE -Wall -Wextra -Wno-misleading-indentation)
include(CTest)
if(BUILD_TESTING)
  add_executable(model-tests tests/model_tests.cpp)
  target_link_libraries(model-tests PRIVATE frotz-model)
  target_compile_options(model-tests PRIVATE -UNDEBUG)
  add_test(NAME model COMMAND model-tests)
  add_executable(media-tests tests/media_tests.cpp)
  target_link_libraries(media-tests PRIVATE frotz-model)
  target_compile_options(media-tests PRIVATE -UNDEBUG)
  add_test(NAME media COMMAND media-tests)
  add_executable(online-tests tests/online_tests.cpp)
  target_include_directories(online-tests PRIVATE ${JSON_INCLUDE_DIR})
  target_link_libraries(online-tests PRIVATE frotz-online Threads::Threads)
  target_compile_options(online-tests PRIVATE -UNDEBUG)
  add_test(NAME online COMMAND online-tests)
  add_executable(archive-tests tests/archive_tests.cpp)
  target_include_directories(archive-tests PRIVATE ${ARCHIVE_INCLUDE_DIR})
  target_link_libraries(archive-tests PRIVATE frotz-online)
  target_compile_options(archive-tests PRIVATE -UNDEBUG)
  add_test(NAME archive COMMAND archive-tests)
endif()
install(TARGETS frotz-zero RUNTIME DESTINATION bin)
install(FILES packaging/frotz-zero.desktop DESTINATION share/APPLaunch/applications)
install(FILES assets/frotz-zero.png DESTINATION share/APPLaunch/share/images)
install(FILES LICENSE THIRD_PARTY_NOTICES.md README.md DESTINATION share/doc/frotz-zero)
